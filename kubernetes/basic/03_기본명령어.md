# kubectl 기본 명령어

```bash
# pod 만들기
kubectl run my-pod -- image nginx

# pod 조회 (상세)
kubectl get pods -o wide

# pod 실시간 조회 
kubectl get pods -w

# pod 상세 정보 조회 
kubectl describe pod my-pod 

# pod 로그 확인 
kubeclt logs -f my-pod

# pod 포트 포워드
kubectl port-forwad pods/my-pod 8080(로컬포트):80(pod포트)
# localhost:8080으로 접속 가능 

# pod 안으로 진입 
kubectl exec -it pods/my-pod -- /bin/bash 

# pod 안의 container에 진입하는 명령어 예시
kubectl exec -it my-pod -c my-container -- sh

kubectl apply -f [filename].yaml # yaml파일로 실행 

kubectl delete pod my-pod   # pod 삭제 
kubectl delete -f pod.yaml   # 리소스와 오브젝트 동시 삭제 
```

Yaml 파일의 기본 포맷을 명령어로 생성 방법

```bash
kubectl run nginx-pod --image nginx --dry-run=client -o yaml>nginx-pod.yaml

kubectl create deployment my-deployment --image nginx --replicas 3 --dry-run=client -o yaml > rs.yaml

# --dry-run=client   리소스를 실제 생성하지 않고 명령을 테스트 하거나 yaml 파일 생성만 실행 
# -o yaml  출력 형식을 yaml로 지정 
# > [파일명].yaml  출력 결과를 yaml  파일로 저장 
```

로그 grep 명령어 

```bash
# 어떤 pod, container 인지 함께 표기  --prefix
# 앱을 로그랩 이라는 라벨이 붙은 파드들의 로그 app=log-app 
kubectl logs -f app=log-lab --prefix  

# 최근 10분 이내의 로그만 가져와라 --since=10m
# 최신 100줄까지만 보여줘라       --tail=100
# | grep -E 'error|warn'     키워드 또는 조건으로 추출 
# | grep -iE 'error|warn'    대소문자 구분없이 키워드 또는 조건으로 추출 
# | grep -F "connection refused"   고정문자열 추출 
# | grep -w "ERROR"     특정 단어 포함된 것만 추출 
# | grep -C2 'error'    에러 단어가 포함된 앞, 뒤 두줄까지 출력 
```

```bash
# pod별 labels 조회 
kubectl get podes --show-labels

# pod별 label의 특정 키의 값 조회 
kubectl get pods -L app

# pod별 label의 특정 키-벨류가 있는 파드 조회 
kubectl get pods -l app=frontend
kubectl get pods --selector="app=frontend"

# pod에 label에 env=test 추가 명령
kubectl label pod [파드이름] env=test 
kubectl label pod [파드이름] env=demo --overwrite  # 수정 
kubectl label pod [파드이름] env- # 삭제.. 키에 -만 붙여주면 됨  

```
